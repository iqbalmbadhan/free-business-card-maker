# Business Card Maker: production settings for Apache / LiteSpeed (cPanel, most shared hosting).
# Rename this file to ".htaccess" and put it in the business-card-maker folder.

Options -Indexes

# ---- private files: dot files (.git, .env), keys, backups and editor files return 404 ----
<IfModule mod_alias.c>
  RedirectMatch 404 /\.(?!well-known/)
  RedirectMatch 404 (?i)\.(pem|p12|key|cer|bak|old|orig|save|swp|log)$
  RedirectMatch 404 ~$
  RedirectMatch 404 (?i)/api/.*\.json$
  RedirectMatch 404 /api/config(\.sample)?\.php$
  RedirectMatch 404 /deploy/
</IfModule>

# ---- compression ----
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css application/javascript application/json application/manifest+json image/svg+xml text/xml
</IfModule>

# ---- serve the pre-compressed page (index.html.gz) when the browser accepts gzip ----
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTP:Accept-Encoding} gzip
  RewriteCond %{REQUEST_FILENAME}index.html.gz -f
  RewriteRule ^$ index.html.gz [L]
  RewriteCond %{HTTP:Accept-Encoding} gzip
  RewriteCond %{REQUEST_FILENAME}.gz -f
  RewriteRule ^index\.html$ index.html.gz [L]
  <FilesMatch "\.html\.gz$">
    ForceType text/html
    Header set Content-Encoding gzip
    Header append Vary Accept-Encoding
    SetEnv no-gzip 1
  </FilesMatch>
</IfModule>

# ---- caching and security headers ----
<IfModule mod_headers.c>
  <FilesMatch "^(index\.html(\.gz)?)?$">
    Header set Cache-Control "public, max-age=0, s-maxage=600, stale-while-revalidate=86400"
    Header always set X-Frame-Options "SAMEORIGIN"
    Header always set Content-Security-Policy "frame-ancestors 'self'; base-uri 'none'; object-src 'none'; form-action 'none'"
    Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=(), usb=()"
  </FilesMatch>
  <Files "sw.js">
    Header set Cache-Control "no-cache"
  </Files>
  <FilesMatch "\.(png|json|ico|xml)$">
    Header set Cache-Control "public, max-age=604800, stale-while-revalidate=86400"
  </FilesMatch>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Strict-Transport-Security "max-age=31536000" "expr=%{HTTPS} == 'on'"
</IfModule>
